unsigned int CRC_16_Calc(unsigned char *ptr, int count)
{
unsigned int  crc, i;
unsigned int Temp;
crc = 0;
while(--count >= 0)
	{
	//Temp=(int)*ptr++ << 8;
	Temp=(int)*ptr;
	Temp<<=8;
	ptr++;
	crc = crc ^ Temp;
	for(i = 0; i < 8; ++i)
			{
			if(crc & 0x8000)
				{
				crc <<= 1 ;
				crc ^= 0x1021;
				}
			else
				crc <<= 1;
				}
	}
	return (crc & 0xFFFF);
}
